int foo(int a, double b, int c, int p){    
    // n, x, groups, size = a, b, c, p
    a = a/p; // a = n*p/p = n
    // 3 sequential loops dependent on n
    for (int i = 0; i < a; ++i)
        c = i + c;  
    
    for (int i = 0; i < a; ++i){
        c = i + c;  
    }

    for (int i = 0; i < n; ++i){c = i + c;}

    a = a + c; // Since a is correlated to n, the return is also correlated to that
    return a;
}

// ---------------------------------------
// main
// ---------------------------------------        
int main(int argc, char* argv[]) {
    
    MPI_Init(&argc, &argv);
    int rank, size;
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);
    double x = 1.0;
    int n = 5;
    int groups = 2;   

    std::vector<double> numbers(n); 

    //MPI Bcast: O(1)
    MPI_Bcast(&n, 1, MPI_INT, 0, MPI_COMM_WORLD);
    
    std::vector<double> numbers(n);
    if (rank == 0)
    {
        for (int i = 0; i < n; ++i)
            numbers[i] = i + 1;  
    }
    //MPI Bcast: O(n = n*p)
    MPI_Bcast(numbers.data(), n, MPI_DOUBLE, 0, MPI_COMM_WORLD);

    // Call functions   
    int T; // T will be correlated to n
    T = foo(n, x, groups, size); 

    // Loop with performance n^2 (T is correlated to n)
    for (int i = 0; i < T; ++i)
        for (int j = 0; j < T; ++i)
            x = j + 1;   
    // Loop with performance n^3 (T is correlated to n)
    for (int i = 0; i < T; ++i){
        for (int j = 0; j < T; ++j){
            for (int k = 0; k < T; ++k){
                x = j + k;  
            }
        }
    }

    MPI_Finalize();
    return 0;
}
